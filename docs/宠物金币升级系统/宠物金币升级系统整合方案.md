**金币计算公式（来自 Habitica 服务端 common/）：**

```javascript
// 参考: habitica/website/common/script/libs/taskHelper.js
// 任务完成得到的金币 = 基础值 × 难度系数 × 连击加成
baseGold = 1.0 + (task.value * 0.1)  // task.value 从 -47.27 到 +21.27 动态变化
goldEarned = baseGold * DIFFICULTY_FACTORS[task.priority]
           // priority: 0.1(简单) / 1.0(普通) / 1.5(难) / 2.0(困难)
```

**移植到 Flutter 的等效实现：**

```dart
// lib/systems/coin_system.dart
class CoinSystem {
  static double calculateReward({
    required double taskValue,  // 任务历史完成率决定的value
    required double priority,   // 0.1 / 1.0 / 1.5 / 2.0
  }) {
    double baseGold = 1.0 + (taskValue * 0.1).clamp(-1.0, 2.0);
    return (baseGold * priority).clamp(0.0, 10.0);
  }
}
```

**升级公式（来自 Habitica common/）：**

```javascript
// 升级所需总经验（第 n 级）
expToLevel(n) = Math.floor(n * 2 * (Math.pow(n, 0.9) / 4.0) + 10)
// 示例：1级→2级需15exp，10级→11级需166exp，50级→51级需2168exp
```

**移植到 Dart：**

```dart
// lib/systems/level_system.dart
class LevelSystem {
  static int expRequiredForLevel(int level) {
    return ((level * 2 * (pow(level, 0.9) / 4.0)) + 10).floor();
  }

  static LevelUpResult checkLevelUp(UserStats stats, double gainedExp) {
    double newExp = stats.exp + gainedExp;
    int newLevel = stats.level;
    while (newExp >= expRequiredForLevel(newLevel)) {
      newExp -= expRequiredForLevel(newLevel);
      newLevel++;
    }
    return LevelUpResult(
      newLevel: newLevel,
      newExp: newExp,
      didLevelUp: newLevel > stats.level,
    );
  }
}
```

### 4.4 宠物成长系统

```dart
// lib/systems/pet_system.dart
class PetSystem {
  static const Map<String, List<String>> petFoodPreference = {
    'Wolf': ['Meat', 'Salmon', 'Chocolate'],
    'BearCub': ['Honey', 'Chocolate', 'Strawberry'],
    'Dragon': ['Any'],  // 吃任何食物，但喜好食物效率x2
    // ...
  };

  static PetFeedResult feedPet({
    required String petKey,
    required String food,
    required int currentValue,
  }) {
    int gain = petFoodPreference[petKey.split('-')[0]]?.contains(food) == true
        ? 5 : 2;   // 喜好食物+5，其他+2（参考Habitica逻辑）
    int newValue = (currentValue + gain).clamp(0, 50);
    bool didEvolve = newValue >= 50;
    return PetFeedResult(newValue: newValue, didEvolveToMount: didEvolve);
  }

  static String? rollPetDrop(Random rng) {
    // 完成任务有5%概率掉落宠物蛋/孵化药水
    if (rng.nextDouble() > 0.05) return null;
    final eggs = ['Wolf', 'BearCub', 'TigerCub', 'Dragon', 'FlyingPig'];
    final potions = ['Base', 'White', 'Desert', 'Golden', 'Zombie'];
    String type = rng.nextBool() ? 'egg' : 'potion';
    if (type == 'egg') return 'egg:${eggs[rng.nextInt(eggs.length)]}';
    return 'potion:${potions[rng.nextInt(potions.length)]}';
  }
}
```

## 五、完成任务时的完整奖励流程

```
孩子点击"完成任务"
        ↓
TaskService.completeTask(chore, child)
        ↓
1. CoinSystem.calculateReward()        → 计算金币
2. LevelSystem.checkLevelUp()          → 计算经验+升级
3. PetSystem.rollPetDrop()             → 5%概率掉落宠物素材
        ↓
4. 更新 UserStats（写入本地JSON）
        ↓
5. 显示奖励弹窗
   ├── "+ 2.5 金币 🪙"
   ├── "+ 15 EXP ⭐"
   ├── （如有）"升级了！第 3 级 🎉"
   └── （如有）"获得：🥚 Wolf 蛋！"
```
